cmake_minimum_required(VERSION 3.12)
project(EmojiDifferenceGame VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(puzzle_generator src/PuzzleGenerator.cpp)
target_include_directories(puzzle_generator PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_executable(emoji_game src/main.cpp)
target_link_libraries(emoji_game PRIVATE puzzle_generator)
add_executable(generator_tests tests/test_generator.cpp)
target_link_libraries(generator_tests PRIVATE puzzle_generator)
enable_testing()
add_test(NAME generator_tests COMMAND generator_tests)
